level_init = function(level_number)
  if level_number == 1 then
    level = Chispita_addMap("overworld/overworld")
    sky = Chispita_addMap("overworld/overworld_background", 0, 0, physics_off)
    space_right = Chispita_addMap("overworld/overworld_outerspace", 1000, 0, physics_off)
    space_right.tile = 1
    space_right.tile_across = 0
    space_left = Chispita_addMap("overworld/overworld_outerspace", -1000, 0, physics_off)
    space_left.tile = 1
    space_left.tile_across = 0
    sky.tile = 1
    sky.parallax = 0.5
    chispita_mode = "platformer"
    level.bounciness = 1
    Camera.zoom = .7
  end
end
